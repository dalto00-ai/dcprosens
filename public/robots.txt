# /dashboard, /login and /embed are deliberately NOT disallowed here.
# They each carry <meta name="robots" content="noindex, nofollow">, and a
# Disallow rule would stop Google from ever fetching the page to read that
# tag — leaving the URL eligible to appear in results as a bare, snippet-less
# "blocked by robots.txt" entry that can never be removed. Letting the
# crawler in is what actually gets them dropped from the index.

# Server-side endpoints — no HTML, nothing to index.
Disallow: /api/
